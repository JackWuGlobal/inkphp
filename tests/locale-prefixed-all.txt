# Language in the URL path with EVERY language prefixed (locale_default_prefixed = true): a bare URL redirects
# (302, Vary: Accept-Language) to the visitor's best language. In-process requests send no Accept-Language,
# so they go to the default (first) language.
@pages tests/fixtures/pages
@config locale_url=prefix locales=en,fr locale_default_prefixed=true

GET     /                     -> 302 /en
GET     /about                -> 302 /en/about
GET     /about?x=1            -> 302 /en/about?x=1
POST    /form                 -> 307 /en/form                  # non-GET keeps its method
GET     /en                   -> 200 pages/index.php []
GET     /en/about             -> 200 pages/about.php []
GET     /fr/about             -> 200 pages/about.php []
GET     /fr/product/42        -> 200 pages/product.php ["42"]
