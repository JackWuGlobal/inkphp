# config locale_exempt: path prefixes outside the language system (admin, login, APIs): one URL, one language.
@pages tests/fixtures/pages
@config locale_url=prefix locales=en,fr locale_default_prefixed=true locale_exempt=/admin,/account,/login

GET     /admin                -> 302 /login?next=%2Fadmin      # no Accept-Language redirect; login is exempt too
GET     /admin/users          -> 302 /login?next=%2Fadmin%2Fusers
GET     /account              -> 302 /login?next=%2Faccount
GET     /fr/admin             -> 301 /admin                    # a language prefix on an exempt URL is dropped
GET     /en/admin/users?x=1   -> 301 /admin/users?x=1
GET     /about                -> 302 /en/about                 # everything else still has languages
GET     /fr/about             -> 200 pages/about.php []
GET     /administrator        -> 302 /en/administrator         # '/admin' matches /admin and /admin/…, not /administrator
