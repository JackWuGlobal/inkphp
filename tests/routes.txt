# Kernel routing table (docs/SPEC.md §4), checked against the fixture pages in tests/fixtures/pages/.
#   bin/ink test                      in-process: canonical redirect + resolver + each page's leading declarations
#   bin/ink test tests/routes.txt --http=http://127.0.0.1:8000
#                                     the same lines against a server whose pages/ is a copy of the fixtures
# Line format: METHOD /raw/path -> STATUS [page file (200) | Location (3xx)] [params as JSON]   # comment
# "pages/…" means a file in the pages directory in use. Paths are used as written; over HTTP only bytes outside
# printable ASCII are percent-encoded first (as browsers do: php -S drops raw UTF-8 request lines).
@pages tests/fixtures/pages
@config locale_url=false                            # these lines test routing alone; languages: tests/locale-*.txt

# --- the root page, and the "index" segment -------------------------------------------------------
GET     /                                  -> 200 pages/index.php []
HEAD    /                                  -> 200 pages/index.php []
GET     /?lang=en                          -> 200 pages/index.php []
GET     /hello                             -> 200 pages/index.php ["hello"]   # the fallback page takes the segments
GET     /0                                 -> 200 pages/index.php ["0"]       # '0' is a real param (0.x dropped it)
GET     /hello/world                       -> 404                             # index.php accepts at most 1 param
GET     /index                             -> 301 /
GET     /index.php                         -> 301 /
GET     /index/                            -> 301 /
GET     /index.php?a=1&b=2                 -> 301 /?a=1&b=2
GET     /index/5                           -> 301 /5
GET     /index/index                       -> 301 /                           # applied until stable: one hop
GET     /index/index.php                   -> 301 /
POST    /index                             -> 308 /                           # 308 keeps the method and body
GET     /foo/index                         -> 404                             # 'index' is only dropped first

# --- .php is stripped from the LAST segment only, in one hop ----------------------------------------
GET     /about                             -> 200 pages/about.php []
HEAD    /about                             -> 200 pages/about.php []
GET     /about.php                         -> 301 /about
GET     /about.php?x=1&y=%2F               -> 301 /about?x=1&y=%2F            # the raw query is kept as-is
GET     /about.php?q=中文                  -> 301 /about?q=%E4%B8%AD%E6%96%87  # non-ASCII is encoded in Location
GET     /about.php.php                     -> 301 /about
GET     /about.php/                        -> 301 /about
HEAD    /about.php                         -> 301 /about
POST    /about.php                         -> 308 /about
PUT     /about/                            -> 308 /about
GET     /zh/popular.php                    -> 301 /zh/popular
GET     /zh/popular.php/2                  -> 404                             # zh.php gets ["popular.php","2"]
GET     /about.php/5                       -> 404                             # index.php gets 2 params
GET     /about%2Ephp                       -> 301 /about                      # %2E is '.', decoded first (§4.1)
GET     /about.PHP                         -> 200 pages/index.php ["about.PHP"]  # only lower-case .php is stripped
GET     /about/x                           -> 404                             # params() accepts none

# --- slashes, dots, and Location safety ------------------------------------------------------------
GET     //about                            -> 301 /about
GET     ///about//                         -> 301 /about
GET     /about/                            -> 301 /about
GET     /./about                           -> 301 /about
GET     /about/.                           -> 301 /about
GET     //evil.com                         -> 301 /evil.com                   # never protocol-relative
GET     //evil.com/                        -> 301 /evil.com
GET     /index//evil.example               -> 301 /evil.example
GET     /\evil.com                         -> 404                             # '\' never reaches a page
GET     /\evil.com/                        -> 301 /%5Cevil.com                # '\' is encoded in Location
GET     //\evil.com                        -> 301 /%5Cevil.com
GET     /%5Cevil.com                       -> 404
GET     /a/../about                        -> 400
GET     /..                                -> 400
GET     /about/..                          -> 400
GET     /.well-known/../x                  -> 400

# --- percent-encoding: unreserved characters are decoded, other escapes upper-cased (one redirect) -----
GET     /%61bout                           -> 301 /about                      # no second URL for a page
GET     /abou%74                           -> 301 /about
GET     /%61dmin                           -> 301 /admin
GET     /%69ndex/5                         -> 301 /5                          # then the other rules, same hop
GET     /about%2ephp                       -> 301 /about
GET     /search/%7e                        -> 301 /search/~
GET     /search/~                          -> 200 pages/search.php ["~"]
GET     /search/%41b%5fC%2D1               -> 301 /search/Ab_C-1
GET     /search/%e4%b8%ad                  -> 301 /search/%E4%B8%AD
POST    /%61bout                           -> 308 /about
GET     /a/%2e%2e/about                    -> 400
GET     /search/%25                        -> 200 pages/search.php ["%"]      # '%' itself stays encoded

# --- percent-decoding: once per segment, UTF-8 only, no separators or controls -----------------------
GET     /search/%E4%B8%AD%E6%96%87         -> 200 pages/search.php ["中文"]
GET     /search/中文                       -> 200 pages/search.php ["中文"]
GET     /search/中文/                      -> 301 /search/%E4%B8%AD%E6%96%87
GET     /search/a%20b                      -> 200 pages/search.php ["a b"]
GET     /search/a+b                        -> 200 pages/search.php ["a+b"]    # '+' is not a space in a path
GET     /search/%2520                      -> 200 pages/search.php ["%20"]    # decoded exactly once
GET     /search/100%25                     -> 200 pages/search.php ["100%"]
GET     /search/0                          -> 200 pages/search.php ["0"]
GET     /search/example.com                -> 200 pages/search.php ["example.com"]  # dots in params are fine
GET     /search/goat.zip                   -> 200 pages/search.php ["goat.zip"]
GET     /search/Ab_C-1                     -> 200 pages/search.php ["Ab_C-1"]       # params keep their case
GET     /search/x.php                      -> 301 /search/x
GET     /search/a%2Fb                      -> 404                             # an encoded '/' never splits
GET     /search/a%2fb                      -> 301 /search/a%2Fb               # other escapes are upper-cased
GET     /search/a%5Cb                      -> 404
GET     /search/a%00b                      -> 404
GET     /search/a%0Ab                      -> 404
GET     /search/%7F                        -> 404
GET     /search/%FF                        -> 404                             # not UTF-8
GET     /search/%C3%28                     -> 404
GET     /search/%2e%2e                     -> 400                             # %2E is '.': this is '..'
GET     /search/%2E                        -> 301 /search                     # and this is '.'
GET     /search/%2E%2E%2E                  -> 301 /search/...
GET     /search/...                        -> 200 pages/search.php ["..."]     # only '.' and '..' are special
GET     /zh%2Fpopular                      -> 404
GET     /search                            -> 404                             # params(1) needs one
GET     /search/a/b                        -> 404
GET     /search?q=a                        -> 404                             # the query is not a path param
