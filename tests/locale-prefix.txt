# Language in the URL path (config locale_url = prefix), default language without a prefix (docs/SPEC.md §4.6).
# In-process against the fixture pages; the kernel strips the language segment, then routes as usual.
# Line format: see tests/routes.txt.
@pages tests/fixtures/pages
@config locale_url=prefix locales=en,fr locale_default_prefixed=false

# --- the default language (en) has no prefix -------------------------------------------------------
GET     /                     -> 200 pages/index.php []
GET     /about                -> 200 pages/about.php []
GET     /en                   -> 301 /                         # the default language is never prefixed
GET     /en/about             -> 301 /about
GET     /en/about?x=1         -> 301 /about?x=1                # the query is kept
POST    /en/form              -> 308 /form                     # non-GET keeps its method

# --- another language: same page file, the prefix is not a param ------------------------------------
GET     /fr                   -> 200 pages/index.php []
GET     /fr/about             -> 200 pages/about.php []
GET     /fr/zh/popular/0      -> 200 pages/zh/popular.php ["0"]   # a later "zh" segment is just a page name
GET     /fr/product/42        -> 200 pages/product.php ["42"]
GET     /fr/about/x           -> 404                           # about.php takes no params, prefix or not
GET     /fr/nope/x/y          -> 404
GET     /fr/about.php         -> 301 /fr/about                 # canonical redirect first, then the language
GET     /fr//about/           -> 301 /fr/about

# --- not a language: an ordinary path -----------------------------------------------------------------
GET     /de/about             -> 404                           # de is not in locales: /de is just an unknown page
GET     /FR/about             -> 404                           # prefixes are exact (lowercase)
GET     /frx/about            -> 404
